const output = document.querySelector<HTMLOutputElement>('#reading-time-result')
const text = document.querySelector('#reading-time-content')?.textContent ?? ''
const words = text.trim().split(/\s+/).filter(Boolean).length
if (output)
  output.textContent = `${words} words · about ${Math.max(1, Math.ceil(words / 200))} minute to read`

## Behavior and accessibility

The default script type is module and the default teleport destination is body. Local entries are resolved from the MDX source and bundled; external URLs remain external script URLs. Keep IDs and event bindings scoped when a page has several independent enhancements.

<ApiProperty anchor="page-script-prop-teleport" name="teleport" valueType="string" defaultValue="body">
  <p>DOM selector receiving the script.</p>
</ApiProperty>

<ApiProperty anchor="page-script-prop-src" name="src" valueType="string" defaultValue="required">
  <p>Browser entry point. A relative path resolves from the current MDX page. SSG records and bundles local scripts.</p>
</ApiProperty>

<ApiProperty anchor="page-script-prop-sourceRelPath" name="sourceRelPath" valueType="string" defaultValue="detected from the current page">
  <p>Optional source-resolution override. Omit it for ordinary colocated MDX and TypeScript files.</p>
</ApiProperty>

<ApiProperty anchor="page-script-prop-type" name="type" valueType="string" defaultValue="module">
  <p>Native script type. Module scripts have browser module loading semantics.</p>
</ApiProperty>

<ApiProperty anchor="page-script-prop-async" name="async" valueType="unknown" defaultValue="not set">
  <p>Native async attribute. Use only when the entry point is independent of execution order.</p>
</ApiProperty>

<ApiProperty anchor="page-script-prop-defer" name="defer" valueType="unknown" defaultValue="not set">
  <p>Native defer attribute. Modules already defer by default.</p>
</ApiProperty>

<ApiProperty anchor="page-script-prop-integrity" name="integrity" valueType="string" defaultValue="not set">
  <p>Subresource integrity metadata for a resource whose bytes you control.</p>
</ApiProperty>

<ApiProperty anchor="page-script-prop-nonce" name="nonce" valueType="string" defaultValue="not set">
  <p>Content Security Policy nonce forwarded to the script.</p>
</ApiProperty>

<ApiProperty anchor="page-script-prop-crossOrigin" name="crossOrigin" valueType="string" defaultValue="not set">
  <p>Native crossorigin request setting.</p>
</ApiProperty>

<ApiProperty anchor="page-script-prop-referrerPolicy" name="referrerPolicy" valueType="string" defaultValue="not set">
  <p>Native script referrerpolicy value.</p>
</ApiProperty>

<ApiProperty anchor="page-script-prop-noModule" name="noModule" valueType="unknown" defaultValue="not set">
  <p>Native nomodule attribute for scripts intended for non-module browsers.</p>
</ApiProperty>
