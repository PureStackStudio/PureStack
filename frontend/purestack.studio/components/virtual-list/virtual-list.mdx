```typescript
import {
  defineBadgeComponents,
  defineButtonComponents,
  defineFlexComponents,
  defineFormComponents,
  defineVirtualListComponents,
} from '@purestack/ts-components'

import {
  createApp,
  defineComponent,
  html,
  type RefOrValue,
  type SRef,
  sref,
} from 'regor'

export interface ExampleRecord {
  id: number
  title: string
  state: string
  detail: string
}
export interface VirtualListRow {
  item: RefOrValue<ExampleRecord>
  index: RefOrValue<number>
}
const rowTemplate = html`<div class="px-3 py-2 bb-1 b-subtle" style="height: 52px; box-sizing: border-box">
  <Flex align="center" justify="between">
    <strong>{{ item.title }}</strong>
    <Badge tone="info">{{ item.state }}</Badge>
  </Flex>
</div>`
const rowComponent = defineComponent<VirtualListRow>(rowTemplate, {
  props: ['item', 'index'],
})

export interface VirtualListExample {
  records: SRef<ExampleRecord[]>
  restore: () => void
  clear: () => void
}

const virtualListExampleTemplate = html`<Flex direction="column">
  <Flex wrap="true">
    <Btn @click="restore">Load 1,000 records</Btn>
    <Btn variant="outline" @click="clear">Empty list</Btn>
    <span role="status">{{ records.length }} records</span>
  </Flex>
  <VirtualList
    :items="records"
    height="320"
    itemHeight="52"
    overscan="4"
    rowComponent="VirtualListRow"
    tabindex="0"
    aria-label="Example change history"
  />
  <FormStatus r-if="records.length === 0">
    No records. Load the example data to continue.
  </FormStatus>
</Flex>`

function createVirtualListExample(): VirtualListExample {
  const initial: ExampleRecord[] = Array.from({ length: 1000 }, (_, index) => ({
    id: index + 1,
    title: `Change ${index + 1}`,
    state: index % 3 === 0 ? 'In review' : 'Ready',
    detail:
      index % 3 === 0
        ? 'This change includes a longer explanation of the component behavior, its theme treatment and the verification needed before release.'
        : 'A focused improvement to the documentation.',
  }))
  const records = sref(initial)
  return {
    records,
    restore: () => records([...initial]),
    clear: () => records([]),
  }
}

const component = defineComponent<VirtualListExample>(
  virtualListExampleTemplate,
  {
    context: createVirtualListExample,
  },
)

createApp(
  {
    components: {
      VirtualListExample: component,
      VirtualListRow: rowComponent,
      ...defineVirtualListComponents(),
      ...defineFlexComponents(),
      ...defineBadgeComponents(),
      ...defineButtonComponents(),
      ...defineFormComponents(),
    },
  },
  { selector: 'app#virtual-list-demo', template: html`<VirtualListExample />` },
)
```

## Behavior and accessibility

The component calculates offsets from itemHeight. Wrapping content or extra vertical padding that exceeds that height breaks alignment. Virtualized off-screen content is absent from the DOM, so browser find, printing and assistive navigation cannot access the whole dataset at once. Offer a non-virtualized export or paginated view when complete access is required. The focusable viewport supports keyboard scrolling.

<ApiProperty anchor="virtual-list-prop-items" name="items" valueType="RefOrValue&lt;unknown[]&gt;" defaultValue="[]">
  <p>Source array. Only the visible window and overscan rows are mounted. Keep durable row state in application data.</p>
</ApiProperty>

<ApiProperty anchor="virtual-list-prop-height" name="height" valueType="RefOrValue&lt;number | string&gt;" defaultValue="560">
  <p>Viewport height in pixels. A positive number or numeric string is accepted. Invalid and non-positive values use the default.</p>
</ApiProperty>

<ApiProperty anchor="virtual-list-prop-itemHeight" name="itemHeight" valueType="RefOrValue&lt;number | string&gt;" defaultValue="44">
  <p>Fixed row height in pixels. The rendered row, including padding and borders, must fit this exact height.</p>
</ApiProperty>

<ApiProperty anchor="virtual-list-prop-overscan" name="overscan" valueType="RefOrValue&lt;number | string&gt;" defaultValue="6">
  <p>Extra rows rendered around the visible window. Positive numeric values are truncated to an integer; zero falls back to 6.</p>
</ApiProperty>

<ApiProperty anchor="virtual-list-prop-rowComponent" name="rowComponent" valueType="RefOrValue&lt;string&gt;" defaultValue="div">
  <p>Registered row component name. Receives item and absolute zero-based index. Provide a typed component that renders the item; the fallback does not format data for you.</p>
</ApiProperty>

### Composition

<div class="api-contract">
<p>No slots. Register defineVirtualListComponents and each named row/header/footer component. Rows receive item and index; they may compose ordinary components. State that must survive scrolling belongs outside the mounted row.</p>
</div>
