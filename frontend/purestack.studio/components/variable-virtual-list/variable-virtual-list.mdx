```typescript
import {
  defineBadgeComponents,
  defineButtonComponents,
  defineFlexComponents,
  defineFormComponents,
  defineVirtualListComponents,
} from '@purestack/ts-components'

import {
  createApp,
  defineComponent,
  html,
  type RefOrValue,
  type SRef,
  sref,
} from 'regor'

export interface ExampleRecord {
  id: number
  title: string
  state: string
  detail: string
}
export interface VariableVirtualListRow {
  item: RefOrValue<ExampleRecord>
  index: RefOrValue<number>
}
const rowTemplate = html`<div class="px-3 py-2 bb-1 b-subtle">
  <Flex align="center" justify="between">
    <strong>{{ item.title }}</strong>
    <Badge tone="info">{{ item.state }}</Badge>
  </Flex>
  <p class="mb-0">{{ item.detail }}</p>
</div>`
const rowComponent = defineComponent<VariableVirtualListRow>(rowTemplate, {
  props: ['item', 'index'],
})

export interface VariableVirtualListExample {
  records: SRef<ExampleRecord[]>
  restore: () => void
  clear: () => void
}

const variableVirtualListExampleTemplate = html`<Flex direction="column">
  <Flex wrap="true">
    <Btn @click="restore">Load 1,000 records</Btn>
    <Btn variant="outline" @click="clear">Empty list</Btn>
    <span role="status">{{ records.length }} records</span>
  </Flex>
  <VariableVirtualList
    :items="records"
    height="320"
    estimateHeight="88"
    overscan="4"
    rowComponent="VariableVirtualListRow"
    tabindex="0"
    aria-label="Example change history"
  />
  <FormStatus r-if="records.length === 0">
    No records. Load the example data to continue.
  </FormStatus>
</Flex>`

function createVariableVirtualListExample(): VariableVirtualListExample {
  const initial: ExampleRecord[] = Array.from({ length: 1000 }, (_, index) => ({
    id: index + 1,
    title: `Change ${index + 1}`,
    state: index % 3 === 0 ? 'In review' : 'Ready',
    detail:
      index % 3 === 0
        ? 'This change includes a longer explanation of the component behavior, its theme treatment and the verification needed before release.'
        : 'A focused improvement to the documentation.',
  }))
  const records = sref(initial)
  return {
    records,
    restore: () => records([...initial]),
    clear: () => records([]),
  }
}

const component = defineComponent<VariableVirtualListExample>(
  variableVirtualListExampleTemplate,
  {
    context: createVariableVirtualListExample,
  },
)

createApp(
  {
    components: {
      VariableVirtualListExample: component,
      VariableVirtualListRow: rowComponent,
      ...defineVirtualListComponents(),
      ...defineFlexComponents(),
      ...defineBadgeComponents(),
      ...defineButtonComponents(),
      ...defineFormComponents(),
    },
  },
  {
    selector: 'app#variable-virtual-list-demo',
    template: html`<VariableVirtualListExample />`,
  },
)
```

## Behavior and accessibility

ResizeObserver measures rendered rows and updates the estimated total height. The scrollbar can adjust as measurements replace estimates. Choose an estimate close to typical row height. Virtualized off-screen content is absent from the DOM, so browser find, printing and assistive navigation cannot access the whole dataset at once. Offer a non-virtualized export or paginated view when complete access is required. The focusable viewport supports keyboard scrolling.

<ApiProperty anchor="variable-virtual-list-prop-items" name="items" valueType="RefOrValue&lt;unknown[]&gt;" defaultValue="[]">
  <p>Source array. Only the visible window and overscan rows are mounted. Keep durable row state in application data.</p>
</ApiProperty>

<ApiProperty anchor="variable-virtual-list-prop-height" name="height" valueType="RefOrValue&lt;number | string&gt;" defaultValue="560">
  <p>Viewport height in pixels. A positive number or numeric string is accepted. Invalid and non-positive values use the default.</p>
</ApiProperty>

<ApiProperty anchor="variable-virtual-list-prop-estimateHeight" name="estimateHeight" valueType="RefOrValue&lt;number | string&gt;" defaultValue="56">
  <p>Initial row height estimate in pixels. ResizeObserver measurements replace estimates as rows enter the viewport.</p>
</ApiProperty>

<ApiProperty anchor="variable-virtual-list-prop-overscan" name="overscan" valueType="RefOrValue&lt;number | string&gt;" defaultValue="6">
  <p>Extra rows rendered around the visible window. Positive numeric values are truncated to an integer; zero falls back to 6.</p>
</ApiProperty>

<ApiProperty anchor="variable-virtual-list-prop-rowComponent" name="rowComponent" valueType="RefOrValue&lt;string&gt;" defaultValue="div">
  <p>Registered row component name. Receives item and absolute zero-based index. Provide a typed component that renders the item; the fallback does not format data for you.</p>
</ApiProperty>

### Composition

<div class="api-contract">
<p>No slots. Register defineVirtualListComponents and each named row/header/footer component. Rows receive item and index; they may compose ordinary components. State that must survive scrolling belongs outside the mounted row.</p>
</div>
