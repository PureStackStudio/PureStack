```typescript
function createLinkPlayground(): LinkPlayground {
  const tone = ref<SemanticTone>('accent')
  const variant = ref<ComponentVariant>('solid')
  const size = ref<BtnSize>('md')
  const href = ref('#link-destination')
  const target = ref('_self')
  const rel = ref('')
  const icon = ref('tabler:arrow-right')
  const position = ref<BtnIconPosition>('end')
  const iconOnly = ref(false)
  const linkLabel = ref('Explore the destination')
  const resolvedIcon = computed(() => (icon() === 'none' ? '' : icon()))
  const resolvedIconOnly = computed(() => iconOnly() && resolvedIcon() !== '')
  const effectiveRel = computed(
    () => rel() || (target() === '_blank' ? 'noopener noreferrer' : 'not set'),
  )
  const options = (values: string[]): FormSelectOption[] =>
    values.map((value) => ({ label: value, value }))
  return {
    tone,
    variant,
    size,
    href,
    target,
    rel,
    icon,
    position,
    iconOnly,
    linkLabel,
    resolvedIcon,
    resolvedIconOnly,
    effectiveRel,
    tones: options(SEMANTIC_TONES),
    variants: options([
      'solid',
      'surface',
      'surfaceAlt',
      'spotlight',
      'glass',
      'flat',
      'flatAlt',
      'flatSolid',
      'outlineFill',
      'outline',
      'subtle',
      'subtleBtn',
      'link',
      'sheen',
      'underline',
      'rail',
      'bracket',
      'none',
    ]),
    sizes: options(['sm', 'md', 'lg']),
    destinations: [
      { label: 'This page’s destination', value: '#link-destination' },
      { label: 'Buttons documentation', value: '/components/actions/buttons/' },
      { label: 'Badge documentation', value: '/components/actions/badge/' },
    ],
    targets: [
      { label: 'Current tab (_self)', value: '_self' },
      { label: 'New tab (_blank)', value: '_blank' },
    ],
    relationships: [
      { label: 'Automatic', value: '' },
      {
        label: 'nofollow + noopener + noreferrer',
        value: 'nofollow noopener noreferrer',
      },
    ],
    icons: options(['none', 'tabler:arrow-right', 'tabler:arrow-up-right']),
    positions: options(['start', 'end']),
    reset: () => {
      icon('tabler:arrow-right')
      iconOnly(false)
      position('end')
      tone('accent')
      variant('solid')
      size('md')
      href('#link-destination')
      target('_self')
      rel('')
      linkLabel('Explore the destination')
    },
  }
}

const icons: Record<string, string> = {
  'lucide:chevron-down': lucide_chevron_down,
  'tabler:arrow-right': tabler_arrow_right,

)
```

## Accessibility

- Give every link a meaningful destination and a name that makes sense in a list of links. Prefer “Read the button guide” to “Click here”.
- A link with an `href` is reachable with Tab and activated with Enter. Browser features such as copying its address and opening it in another tab remain available.
- Let visible text name the link. Supply `ariaLabel` for an icon-only link, or when the visible text needs more context.
- Signal when a destination opens in a new tab. The examples include this information in the label.
- Add `aria-current="page"` yourself for the current page in a navigation region. BtnLink does not detect the current route.
- BtnLink has no disabled state. If navigation is unavailable, present explanatory text or omit the link until its destination is available.

### Navigation

<ApiProperty anchor="link-prop-href" name="href" valueType="string" defaultValue="not set">
  <p>The destination of the anchor. Accepts page paths, absolute URLs, and fragments. During server rendering, PureStack resolves site paths through the current public URL context.</p>
  <div class="api-values"><code>/components/actions/buttons/</code><code>#playground</code><code>https://example.com</code></div>
  <p class="api-note">An omitted or empty value produces an anchor without a destination. Supply a real URL for usable navigation.</p>
```html
<BtnLink href="/components/actions/buttons/">Read the button guide</BtnLink>
```
</ApiProperty>

<ApiProperty anchor="link-prop-target" name="target" valueType="string" defaultValue="not set">
  <p>Selects the browsing context. Without a target, the browser uses its normal navigation behavior. A named frame is also supported.</p>
  <div class="api-values"><code>_self</code><code>_blank</code><code>_parent</code><code>_top</code><code>frame name</code></div>
```html
<BtnLink href="https://github.com/PureStackStudio/PureStack" target="_blank">
  PureStack on GitHub (new tab)
</BtnLink>
```
</ApiProperty>

<ApiProperty anchor="link-prop-rel" name="rel" valueType="string" defaultValue="automatic for _blank">
  <p>Space-separated relationship tokens. An omitted or empty value becomes <code>noopener noreferrer</code> for <code>target="_blank"</code>; otherwise the attribute is omitted.</p>
  <p class="api-note">An explicit value replaces the automatic value; tokens are not merged.</p>
```html
<BtnLink href="https://github.com/PureStackStudio/PureStack" target="_blank" rel="nofollow noopener noreferrer">
  Repository (new tab)
</BtnLink>
```
</ApiProperty>

### Icons and labels

<ApiProperty anchor="link-prop-icon" name="icon" valueType="string" defaultValue="not set">
  <p>An icon name from the configured registry. Empty or omitted values render no icon. Browser applications must register the icons they use.</p>

</ApiProperty>

<ApiProperty anchor="link-prop-icon-position" name="iconPosition" valueType="BtnIconPosition" defaultValue="start">
  <p>Places the icon before or after the visible label. Has no visual effect without an icon.</p>
  <div class="api-values"><code>start</code><code>end</code></div>

</ApiProperty>

<ApiProperty anchor="link-prop-icon-only" name="iconOnly" valueType="boolean" defaultValue="false">
  <p>Hides the visible label and uses a compact icon treatment. Supply both an icon and an accessible name.</p>
```html
<BtnLink href="#playground" icon="tabler:arrow-right" :iconOnly="true" ariaLabel="Open the link playground"/>
```
</ApiProperty>

<ApiProperty anchor="link-prop-aria-label" name="ariaLabel" valueType="string" defaultValue="not set">
  <p>Sets the anchor's <code>aria-label</code>. Omit it when visible text already names the destination. When provided, it becomes the accessible name.</p>
  <p class="api-note">Keep the visible label in the accessible name when both are present, so speech navigation can identify the link.</p>
</ApiProperty>

### Slots and native attributes

<div class="api-contract">
  <h4>Default slot</h4>
  <p>The visible link label. Accepts text and inline markup; keep other interactive controls outside the link. Hidden when <code>iconOnly</code> is true.</p>
  <h4>Native attributes</h4>
  <p>Attributes such as <code>id</code>, <code>class</code>, <code>title</code>, <code>download</code>, and <code>aria-current</code> reach the anchor. Native browser rules still apply to downloads and browsing contexts.</p>
  <h4>Events</h4>
  <p>Native anchor events are available in Regor apps. Navigation needs no handler. If you add <code>@click</code>, preserve the browser's default behavior unless your application intentionally handles the navigation.</p>
