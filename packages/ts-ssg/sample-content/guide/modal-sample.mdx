# Modal Component Samples

`Modal` and `ModalTrigger` provide an accessible, progressively-enhanced dialog system.

## 1. Basic modal

<ModalTrigger
  target="contact-modal"
  label="Open contact modal"
  showClose="true"/>

<Modal id="contact-modal" title="Contact sales" showClose="true">
  <p>
    Share your use case and expected traffic profile. We can recommend the right
    plan and rollout path.
  </p>
  <template name="footer">
    <Flex wrap="true">
      <FormSubmit label="Book a call"/>
      <FormSubmit label="Send details"/>
    </Flex>
  </template>
</Modal>

## 5. Full content override via content slot

<ModalTrigger target="full-content-modal" label="Open full-content modal"/>

<Modal id="full-content-modal" fade="false" slideFrom="left" size="md">
  <template name="content">
    <article class="modal__panel" role="document" tabindex="-1">
      <h2>Fully custom content slot</h2>
      <p>
        This replaces the default modal shell entirely when you need custom
        internal structure.
      </p>
      <Grid columns="1" justifyItems="center">
        <Flex wrap="true">
          <Btn type="button" class="modal-trigger" data-modal-close>
            Approve
          </Btn>
          <Btn type="button" class="modal-trigger" data-modal-close>Cancel</Btn>
        </Flex>
      </Grid>
    </article>
  </template>
</Modal>

## 8. Advanced form modal (nested grids + status)

<ModalTrigger target="advanced-form-modal" label="Open advanced form modal"/>

<Modal
  id="advanced-form-modal"
  title="Create rollout request"
  size="xl"
  fade="true"
  slideFrom="right"
  showClose="true"
>
  <AppForm action="/rollout/request" method="post">
    <Grid columns="1" columnsMd="2">
      <FormInputField
        label="Owner name"
        type="text"
        name="ownerName"
        placeholder="Ada Lovelace"
        autocomplete="name"
        required="true"/>
      <FormInputField
        label="Owner email"
        type="email"
        name="ownerEmail"
        placeholder="ada@analytical.engine"
        autocomplete="email"
        required="true"/>
    </Grid>

    <Grid columns="1" columnsLg="3">
      <FormInputField
        label="Environment"
        type="text"
        name="environment"
        placeholder="production-eu"
        required="true"/>
      <FormInputField
        label="Canary percent"
        type="number"
        name="canaryPercent"
        min="1"
        max="100"
        step="1"
        value="10"/>
      <FormInputField
        label="Expected duration (minutes)"
        type="number"
        name="durationMinutes"
        min="5"
        step="5"
        value="30"/>
    </Grid>

    <Grid columns="1" columnsMd="2" alignItems="start">
      <FormCheck
        label="Notify on-call channel when rollout starts"
        name="notifyOnCall"
        value="yes"
        checked="true"/>
      <FormCheck
        label="I confirm rollback plan is documented"
        name="rollbackConfirmed"
        value="true"
        checked="true"/>
    </Grid>

    <FormStatus tone="info">
      Static demo: submission is for layout and interaction preview only.
    </FormStatus>
  </AppForm>
  <template name="footer">
    <Flex wrap="true">
      <Btn type="button" class="modal-trigger" data-modal-close>Cancel</Btn>
      <Btn type="button" class="modal-trigger">Save draft</Btn>
      <Btn type="button" class="modal-trigger">Submit request</Btn>
    </Flex>
  </template>
</Modal>

## 9. Nested modal flow (modal opens another modal)

<ModalTrigger target="parent-modal" label="Open parent modal"/>

<Modal
  id="parent-modal"
  title="Primary approval"
  size="lg"
  fade="true"
  slideFrom="right"
  showClose="true"
>
  <p>
    This primary modal can launch a secondary confirmation modal for destructive
    or high-risk actions.
  </p>
  <template name="footer">
    <Flex wrap="true">
      <Btn type="button" class="modal-trigger" data-modal-close>Cancel</Btn>
      <ModalTrigger target="child-modal" label="Continue to confirmation"/>
    </Flex>
  </template>
</Modal>

<Modal
  id="child-modal"
  title="Secondary confirmation"
  size="md"
  fade="true"
  slideFrom="bottom"
  showClose="true"
>
  <AlertBox tone="warning" title="Final confirmation required">
    You are about to apply changes to the production environment.
  </AlertBox>
  <template name="footer">
    <Flex wrap="true">
      <Btn type="button" class="modal-trigger" data-modal-close>Back</Btn>
      <Btn type="button" class="modal-trigger" data-modal-close>
        Confirm rollout
      </Btn>
    </Flex>
  </template>
</Modal>

## 10. Props reference

```html
<Modal
  id="my-modal"
  title="Dialog title"
  tone="neutral|accent|info|success|warning|danger|ghost"
  size="sm|md|lg|xl"
  showClose="true|false"
  fade="true|false"
  slideFrom="none|top|right|bottom|left"
>
  default body content

  <template name="header">...</template>
  <template name="body">...</template>
  <template name="footer">...</template>
  <template name="content">...</template>
</Modal>

<ModalTrigger target="my-modal" label="Open" />
```
