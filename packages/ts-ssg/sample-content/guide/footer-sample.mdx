## 4. Quick prop cookbook

```html
<SiteFooter
  teleport="selector"
  eyebrow="Optional eyebrow"
  title="Footer title"
  tagline="Optional long tagline"
  tone="default|accent|neutral"
  variant="default|minimal|feature"
  ctaLabel="Call to action"
  ctaHref="/start/"
  ctaTarget="_blank"
  newsletter="true|false"
  newsletterTitle="Newsletter title"
  newsletterBody="Newsletter helper text"
  newsletterAction="/subscribe"
  newsletterMethod="get|post"
  newsletterButtonLabel="Subscribe"
>
  <p>Default slot content</p>

  <template name="status">
    <span>Status badge</span>
  </template>

  <template name="columns">
    <FooterColumn title="Column title" compact="true|false">
      <FooterLink
        href="/path/"
        label="Link"
        variant="default|muted|strong"
        icon="external|arrow"
        target="_blank"
      />
    </FooterColumn>
  </template>

  <template name="legal">
    <a href="/privacy/">Privacy</a>
  </template>

  <template name="social">
    <FooterSocial href="https://github.com" label="GitHub" target="_blank" />
  </template>
</SiteFooter>
```
